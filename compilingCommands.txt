//32 bit nasm assembler
nasm -felf32 boot.asm -o boot.o

//c compiler
i686-elf-gcc -c kernel.c -o kernel.o -std=gnu99 -ffreestanding -O2 -Wall -Wextra

//Linker
i686-elf-gcc -T linker.ld -o Blueberry -ffreestanding -O2 -nostdlib boot.o kernel.o -lgcc

//Checking Multiboot
grub-file --is-x86-multiboot myos
